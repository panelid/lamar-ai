// Worker: lamar-ai (Phase 1 MVP) // IMPORTANT: Pastikan kamu sudah bind KV namespaces: // - CV_USERS (KV) -> untuk menyimpan user { passwordHash, email } // - SESSIONS (KV) -> untuk menyimpan session tokens // Also ensure env.AI is available. addEventListener("fetch", event => { event.respondWith(handleRequest(event.request, event)); }); function jsonResponse(obj, status = 200) { return new Response(JSON.stringify(obj), { status, headers: { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET,POST,OPTIONS", "Access-Control-Allow-Headers": "Content-Type, Authorization" } }); } function textResponse(text, status = 200) { return new Response(text, { status, headers: { "Content-Type": "text/plain; charset=utf-8", "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET,POST,OPTIONS", "Access-Control-Allow-Headers": "Content-Type, Authorization" } }); } async function hashPassword(password) { const enc = new TextEncoder(); const data = enc.encode(password); const hashBuf = await crypto.subtle.digest("SHA-256", data); const hashArray = Array.from(new Uint8Array(hashBuf)); return hashArray.map(b => b.toString(16).padStart(2, "0")).join(""); } function makeToken() { const arr = new Uint8Array(24); crypto.getRandomValues(arr); return Array.from(arr).map(b => b.toString(16).padStart(2, "0")).join(""); } async function handleRequest(request, event) { if (request.method === "OPTIONS") { return new Response(null, { status: 204, headers: { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET,POST,OPTIONS", "Access-Control-Allow-Headers": "Content-Type, Authorization" } }); } const url = new URL(request.url); const path = url.pathname.replace(/\/+$/, "") || "/"; try { if (path === "/" && request.method === "GET") { // Simple message / health check return textResponse("lamar-ai worker: ready"); } // Signup: create user (slug + password + email) if (path === "/signup" && request.method === "POST") { const body = await request.json(); const { slug, password, email } = body || {}; if (!slug || !password) return jsonResponse({ error: "slug and password required" }, 400); const existing = await CV_USERS.get(slug); if (existing) return jsonResponse({ error: "slug already exists" }, 409); const pwHash = await hashPassword(password); const data = { password: pwHash, email: email || "" }; await CV_USERS.put(slug, JSON.stringify(data)); return jsonResponse({ ok: true }); } // Login: return token if (path === "/login" && request.method === "POST") { const body = await request.json(); const { slug, password } = body || {}; if (!slug || !password) return jsonResponse({ error: "slug and password required" }, 400); const stored = await CV_USERS.get(slug); if (!stored) return jsonResponse({ error: "invalid credentials" }, 401); const parsed = JSON.parse(stored); const pwHash = await hashPassword(password); if (pwHash !== parsed.password) return jsonResponse({ error: "invalid credentials" }, 401); const token = makeToken(); // store session token for 7 days await SESSIONS.put("sess:" + token, slug, { expirationTtl: 60 * 60 * 24 * 7 }); return jsonResponse({ ok: true, token }); } // Generate improved CV: requires Authorization: Bearer if (path === "/generate" && request.method === "POST") { const auth = request.headers.get("Authorization") || ""; const token = auth.startsWith("Bearer ") ? auth.slice(7) : null; if (!token) return jsonResponse({ error: "missing token" }, 401); const owner = await SESSIONS.get("sess:" + token); if (!owner) return jsonResponse({ error: "invalid/expired token" }, 401); const body = await request.json(); const cvText = body.cvText || ""; const template = body.template || "classic"; if (!cvText || cvText.trim().length < 10) return jsonResponse({ error: "cvText too short" }, 400); // Build prompt for AI const prompt = ` You are a professional CV writer. Input CV (raw) below. Task: Rewrite and create a clean, professional CV formatted text (sections: Name, Contact, Profile Summary, Experience, Education, Skills). Language: Indonesian. Tone: Formal, HR-friendly. Template: ${template} Raw CV: ${cvText} `; // Call Cloudflare Workers AI let aiResult; try { const aiResponse = await env.AI.run("@cf/meta/llama-3-8b-instruct", { prompt, max_output_tokens: 1200 }); // aiResponse can be string or object depending on platform version - normalize: if (typeof aiResponse === "string") { aiResult = aiResponse; } else if (aiResponse?.output?.[0]?.content) { // newer response shape aiResult = aiResponse.output[0].content.map(c => c.text || c?.parts?.[0]).join("\n"); } else if (aiResponse?.[0]?.text) { aiResult = aiResponse[0].text; } else { aiResult = JSON.stringify(aiResponse); } } catch (e) { return jsonResponse({ error: "AI error: " + (e.message || e.toString()) }, 500); } // optional: save a copy (timestamped) try { const key = `cv:${owner}:${Date.now()}`; await CVS.put(key, aiResult); } catch (e) { // if CVS not bound, ignore } return jsonResponse({ ok: true, result: aiResult }); } return jsonResponse({ error: "not found" }, 404); } catch (e) { return jsonResponse({ error: "server error", detail: e.message || String(e) }, 500); } }
